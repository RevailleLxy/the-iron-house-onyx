CREATE DATABASE IF NOT EXISTS ironhouse_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE ironhouse_db;

-- 1. Tabla de Usuarios del Sistema (Roles: Administrador y Recepcionista)
CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre_completo VARCHAR(100) NOT NULL,
    usuario VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    rol ENUM('Administrador', 'Recepcionista') NOT NULL,
    activo TINYINT(1) DEFAULT 1,
    fecha_creacion DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 2. Catálogo de Planes y Precios (Solo modificable por Administrador)
CREATE TABLE cat_planes (
    id_plan INT AUTO_INCREMENT PRIMARY KEY,
    nombre_plan VARCHAR(50) NOT NULL,
    duracion_dias INT NOT NULL,
    es_pareja TINYINT(1) DEFAULT 0,
    precio DECIMAL(10,2) NOT NULL,
    activo TINYINT(1) DEFAULT 1
);

-- 3. Tabla de Socios (Con contacto de emergencia y parentesco)
CREATE TABLE socios (
    id_socio INT AUTO_INCREMENT PRIMARY KEY,
    folio_socio VARCHAR(20) UNIQUE,
    nombre_completo VARCHAR(120) NOT NULL,
    fecha_nacimiento DATE NOT NULL,
    celular VARCHAR(20) NOT NULL,
    correo VARCHAR(100),
    emergencia_nombre VARCHAR(120) NOT NULL,
    emergencia_telefono VARCHAR(20) NOT NULL,
    emergencia_parentesco VARCHAR(50) NOT NULL,
    id_plan_actual INT NULL,
    fecha_inicio_membresia DATE NULL,
    fecha_vencimiento DATE NULL,
    id_socio_pareja INT NULL,
    estatus_registro ENUM('Activo', 'Baja') DEFAULT 'Activo',
    fecha_registro DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_plan_actual) REFERENCES cat_planes(id_plan)
);

-- 4. Tabla de Pagos y Renovaciones (Para recibos PDF y reportes de ingresos)
CREATE TABLE pagos (
    id_pago INT AUTO_INCREMENT PRIMARY KEY,
    folio_recibo VARCHAR(30) NOT NULL UNIQUE,
    id_socio INT NOT NULL,
    id_socio_secundario INT NULL, -- Por si es plan de parejas
    id_plan INT NOT NULL,
    monto DECIMAL(10,2) NOT NULL,
    metodo_pago ENUM('Efectivo', 'Tarjeta', 'Transferencia') NOT NULL,
    fecha_pago DATETIME DEFAULT CURRENT_TIMESTAMP,
    vigencia_inicio DATE NOT NULL,
    vigencia_fin DATE NOT NULL,
    id_usuario_cobro INT NOT NULL,
    FOREIGN KEY (id_socio) REFERENCES socios(id_socio),
    FOREIGN KEY (id_plan) REFERENCES cat_planes(id_plan),
    FOREIGN KEY (id_usuario_cobro) REFERENCES usuarios(id_usuario)
);

-- 5. Bitácora de Actividades (Auditoría por usuario)
CREATE TABLE bitacora (
    id_bitacora INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    modulo VARCHAR(50) NOT NULL,
    accion VARCHAR(255) NOT NULL,
    fecha_hora DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);
